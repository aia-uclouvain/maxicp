package org.maxicp.cp.engine.constraints.scheduling;

import org.maxicp.cp.engine.core.AbstractCPConstraint;
import org.maxicp.cp.engine.core.CPIntervalVar;

/**
 * Constrains interval A to end before or at the start of interval B if both are present.
 *
 * @author Pierre Schaus
 */
public class EndBeforeStart extends AbstractCPConstraint {

    final CPIntervalVar A, B;

    /**
     * Creates a constraint enforcing {@code end(A) <= start(B)} whenever both {@code A} and {@code B} are present.
     *
     * @param A the first interval variable
     * @param B the second interval variable
     */
    public EndBeforeStart(CPIntervalVar A, CPIntervalVar B) {
        super(A.getSolver());
        this.A = A;
        this.B = B;
    }

    @Override
    public void post() {
        if (!A.isAbsent() && !B.isAbsent()) {
            A.propagateOnChange(this);
            B.propagateOnChange(this);
            propagate();
        }
    }

    @Override
    public void propagate() {
        if (A.isPresent()) {
            B.setStartMin(A.endMin());
        }
        if (B.isPresent()) {
            A.setEndMax(B.startMax());
        }
        if (A.isAbsent() || B.isAbsent()) {
            setActive(false);
        }
    }
}
